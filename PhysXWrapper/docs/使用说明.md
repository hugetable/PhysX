## 1. 项目简介

**PhysXWrapper** 是对 NVIDIA PhysX 物理引擎的高级封装库，旨在简化PhysX的使用，提供更加友好的C++接口。

### 1.1 主要特性

- ✅ **简化的API**: 封装复杂的PhysX初始化和配置
- ✅ **丰富的示例**: 基于官方Snippets改编的实用类
- ✅ **完善的文档**: 中英文文档和注释
- ✅ **模块化设计**: 按功能分类，可按需引入
- ✅ **资源管理**: 自动管理PhysX对象生命周期
- ✅ **错误处理**: 友好的错误信息和异常处理

### 1.2 系统要求

**操作系统**:
- Windows 10/11
- Linux (Ubuntu 20.04+, CentOS 7+)
- macOS 10.15+

**编译器**:
- GCC 7.0+
- Clang 6.0+
- MSVC 2017+

**依赖**:
- CMake 3.15+
- CUDA 11.0+ (可选，GPU加速功能)
- Python 3.6+ (构建脚本)

## 2. 编译PhysX核心库

### 2.1 Linux系统

#### 方法一：使用官方脚本（推荐）

```bash
cd /path/to/PhysX/physx
./generate_projects.sh linux

cd compiler/linux-release
make -j8

# 编译Debug版本
cd ../linux-debug
make -j8
```

编译完成后，库文件位于：
```
physx/bin/linux.clang/release/
├── libPhysX_64.so
├── libPhysXCommon_64.so
├── libPhysXCooking_64.so
├── libPhysXFoundation_64.so
├── libPhysXExtensions_static_64.a
└── ...
```

#### 方法二：使用CMake

```bash
cd /path/to/PhysX/physx
mkdir build && cd build

cmake .. \
    -DCMAKE_BUILD_TYPE=Release \
    -DPX_BUILDSNIPPETS=OFF \
    -DPX_BUILDPUBLICSAMPLES=OFF \
    -DPX_GENERATE_STATIC_LIBRARIES=OFF \
    -DPX_ENABLE_GPU_PHYSX=ON

cmake --build . --config Release -j8
```

```powershell
cd C:\path\to\PhysX\physx
```

## 3. 编译PhysXWrapper库

### 3.1 配置CMake

创建 `PhysXWrapper/CMakeLists.txt`:

```cmake
cmake_minimum_required(VERSION 3.15)
project(PhysXWrapper VERSION 1.0.0 LANGUAGES CXX)

# C++ 标准
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# PhysX 路径配置
set(PHYSX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../physx" CACHE PATH "PhysX root directory")
set(PHYSX_INCLUDE_DIR "${PHYSX_ROOT}/include")
set(PHYSX_LIB_DIR "${PHYSX_ROOT}/bin/${PX_PLATFORM}/release")

# 包含目录
include_directories(
    ${CMAKE_CURRENT_SOURCE_DIR}/include
    ${PHYSX_INCLUDE_DIR}
    ${PHYSX_ROOT}/source/foundation/include
)

# 查找PhysX库
find_library(PHYSX_FOUNDATION PhysXFoundation_64 HINTS ${PHYSX_LIB_DIR})
find_library(PHYSX_COMMON PhysXCommon_64 HINTS ${PHYSX_LIB_DIR})
find_library(PHYSX_PHYSICS PhysX_64 HINTS ${PHYSX_LIB_DIR})
find_library(PHYSX_COOKING PhysXCooking_64 HINTS ${PHYSX_LIB_DIR})
find_library(PHYSX_EXTENSIONS PhysXExtensions_static_64 HINTS ${PHYSX_LIB_DIR})

# 源文件
file(GLOB_RECURSE WRAPPER_SOURCES
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp"
)

file(GLOB_RECURSE WRAPPER_HEADERS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.h"
)

# 创建库
add_library(PhysXWrapper SHARED ${WRAPPER_SOURCES} ${WRAPPER_HEADERS})

# 链接PhysX库
target_link_libraries(PhysXWrapper
    ${PHYSX_FOUNDATION}
    ${PHYSX_COMMON}
    ${PHYSX_PHYSICS}
    ${PHYSX_COOKING}
    ${PHYSX_EXTENSIONS}
)

# 安装规则
install(TARGETS PhysXWrapper
    LIBRARY DESTINATION lib
    ARCHIVE DESTINATION lib
    RUNTIME DESTINATION bin
)

install(DIRECTORY include/ DESTINATION include/PhysXWrapper)

# 示例程序（可选）
option(BUILD_EXAMPLES "Build example programs" ON)
if(BUILD_EXAMPLES)
    add_subdirectory(examples)
endif()
```

### 3.2 Linux编译

```bash
cd /path/to/PhysX/PhysXWrapper
mkdir build && cd build

cmake .. \
    -DCMAKE_BUILD_TYPE=Release \
    -DPHYSX_ROOT=/path/to/PhysX/physx

make -j8

# 安装（可选）
sudo make install
```

### 3.3 Windows编译

```powershell
cd C:\path\to\PhysX\PhysXWrapper
mkdir build
cd build

cmake .. -G "Visual Studio 17 2022" -A x64 `
    -DPHYSX_ROOT="C:\path\to\PhysX\physx"

cmake --build . --config Release
```

```

├── libPhysXWrapper.so / PhysXWrapper.dll / libPhysXWrapper.dylib
└── examples/ (如果启用)
```

## 4. 在项目中使用PhysXWrapper

### 4.1 CMake集成

在您的项目 `CMakeLists.txt` 中：

```cmake
cmake_minimum_required(VERSION 3.15)
project(MyPhysicsApp)

set(CMAKE_CXX_STANDARD 17)

# 设置PhysXWrapper路径
set(PHYSXWRAPPER_DIR "/path/to/PhysXWrapper")

# 包含头文件
include_directories(
    ${PHYSXWRAPPER_DIR}/include
    ${PHYSXWRAPPER_DIR}/../physx/include
)

# 链接库目录
link_directories(
    ${PHYSXWRAPPER_DIR}/build
    ${PHYSXWRAPPER_DIR}/../physx/bin/linux.clang/release
)

# 创建可执行文件
add_executable(MyApp main.cpp)

# 链接PhysXWrapper和PhysX库
target_link_libraries(MyApp
    PhysXWrapper
    PhysX_64
    PhysXCommon_64
    PhysXFoundation_64
    PhysXCooking_64
)
```

### 4.2 手动编译链接

#### Linux

```bash
g++ -std=c++17 \
    -I/path/to/PhysXWrapper/include \
    -I/path/to/PhysX/physx/include \
    -L/path/to/PhysXWrapper/build \
    -L/path/to/PhysX/physx/bin/linux.clang/release \
    -o MyApp main.cpp \
    -lPhysXWrapper \
    -lPhysX_64 \
    -lPhysXCommon_64 \
    -lPhysXFoundation_64 \
    -lPhysXCooking_64 \
    -lpthread
```

#### Windows

```powershell
cl /std:c++17 /EHsc `
    /I"C:\path\to\PhysXWrapper\include" `
    /I"C:\path\to\PhysX\physx\include" `
    main.cpp `
    /link `
    /LIBPATH:"C:\path\to\PhysXWrapper\build\Release" `
    /LIBPATH:"C:\path\to\PhysX\physx\bin\win.x86_64.vc142.md\release" `
    PhysXWrapper.lib `
    PhysX_64.lib `
    PhysXCommon_64.lib `
    PhysXFoundation_64.lib `
    PhysXCooking_64.lib
```

### 4.3 运行时库路径配置

#### Linux

```bash
# 临时设置
export LD_LIBRARY_PATH=/path/to/PhysXWrapper/build:/path/to/PhysX/physx/bin/linux.clang/release:$LD_LIBRARY_PATH

# 永久设置（添加到 ~/.bashrc）
echo 'export LD_LIBRARY_PATH=/path/to/PhysXWrapper/build:/path/to/PhysX/physx/bin/linux.clang/release:$LD_LIBRARY_PATH' >> ~/.bashrc
```

#### Windows

将以下目录添加到系统 PATH 环境变量：
- `C:\path\to\PhysXWrapper\build\Release`
- `C:\path\to\PhysX\physx\bin\win.x86_64.vc142.md\release`

## 5. 快速开始示例

### 5.1 最简单的示例

```cpp
#include <PhysXWrapper/Core/PhysXCore.h>
#include <iostream>

int main() {
    // 1. 创建PhysX核心对象
    PhysXCore physx;

    // 2. 初始化
    if (!physx.initialize()) {
        std::cerr << "Failed to initialize PhysX: "
                  << physx.getLastError() << std::endl;
        return 1;
    }

    // 3. 创建一个简单场景
    PhysXCore::SceneConfig sceneConfig;
    sceneConfig.gravity = {0.0f, -9.81f, 0.0f};

    if (!physx.createScene(sceneConfig)) {
        std::cerr << "Failed to create scene" << std::endl;
        return 1;
    }

    // 4. 添加地面（静态平面）
    physx.addGroundPlane(0.0f); // y = 0

    // 5. 添加一个下落的球体
    PhysXCore::RigidBodyConfig ballConfig;
    ballConfig.position = {0.0f, 10.0f, 0.0f};
    ballConfig.shape = PhysXCore::Shape::Sphere;
    ballConfig.radius = 1.0f;
    ballConfig.mass = 1.0f;
    ballConfig.isDynamic = true;

    int ballId = physx.addRigidBody(ballConfig);

    // 6. 运行模拟
    for (int i = 0; i < 100; ++i) {
        physx.update(1.0f / 60.0f); // 60 FPS

        // 获取球体位置
        auto transform = physx.getTransform(ballId);
        std::cout << "Frame " << i << ": Ball position = ("
                  << transform.position.x << ", "
                  << transform.position.y << ", "
                  << transform.position.z << ")" << std::endl;
    }

    // 7. 清理
    physx.cleanup();

    return 0;
}
```

```cpp
public:
    void onContactBegin(int bodyA, int bodyB, const ContactInfo& info) override {
        std::cout << "Contact between " << bodyA << " and " << bodyB << std::endl;

    physx.initialize();

    contactHandler.setCallback(&callback);

        physx.update(deltaTime);
    }
```

```cpp
    return 0;
}
```

### 5.5 使用车辆

```cpp
#include <PhysXWrapper/Vehicle/VehicleFourWheels.h>

int main() {
    PhysXCore physx;
    physx.initialize();
    physx.createScene();

    VehicleFourWheels vehicle(physx.getPhysics());

    VehicleFourWheels::Config vehConfig;
    vehConfig.chassisMass = 1500.0f;
    vehConfig.chassisDimensions = {2.5f, 1.0f, 5.0f};
    vehConfig.wheelRadius = 0.4f;
    vehConfig.wheelWidth = 0.3f;

    vehicle.create(vehConfig);

    // 控制车辆
    while (running) {
        VehicleFourWheels::Controls controls;
        controls.throttle = getThrottleInput(); // 0.0 ~ 1.0
        controls.brake = getBrakeInput();       // 0.0 ~ 1.0
        controls.steering = getSteeringInput(); // -1.0 ~ 1.0

        vehicle.update(controls, deltaTime);
        physx.update(deltaTime);
    }

    return 0;
}
```

## 6. 常见问题

### 6.1 编译问题

**Q: CMake找不到PhysX库**
```
A: 确保设置了正确的PHYSX_ROOT路径，或者手动指定：
   cmake .. -DPHYSX_ROOT=/path/to/PhysX/physx
```

**Q: 链接错误：undefined reference**
```
A: 检查：
   1. 库文件是否存在于指定路径
   2. 链接库的顺序（PhysXWrapper应该在PhysX库之前）
   3. 是否链接了所有必需的库
```

**Q: C++版本不兼容**
```
A: PhysXWrapper需要C++17，确保：
   - GCC 7.0+
   - Clang 6.0+
   - MSVC 2017+
```

### 6.2 运行时问题

**Q: 程序运行时找不到.so/.dll文件**
```
A: 设置运行时库路径：
   Linux: export LD_LIBRARY_PATH=/path/to/libs:$LD_LIBRARY_PATH
   Windows: 添加到系统PATH
   macOS: export DYLD_LIBRARY_PATH=/path/to/libs:$DYLD_LIBRARY_PATH
```

**Q: GPU功能无法使用**
```
A: 检查：
   1. 是否有支持CUDA的NVIDIA显卡
   2. 是否安装了CUDA驱动（11.0+）
   3. PhysX是否使用PX_ENABLE_GPU_PHYSX=ON编译
   4. 场景创建时是否设置useGPU=true
```

**Q: 物理模拟不准确**
```
A: 调整：
   1. 时间步长（建议1/60秒）
   2. 求解器迭代次数
   3. 质量比例（避免过大或过小的质量差异）
   4. 碰撞容差
```

### 6.3 性能问题

**Q: 帧率很低**
```
A: 优化建议：
   1. 减少物体数量
   2. 简化碰撞网格
   3. 使用凸包代替三角网格
   4. 启用睡眠机制
   5. 使用GPU加速（如果可用）
   6. 调整广相位算法
```

**Q: 内存占用很高**
```
A: 检查：
   1. 是否正确释放了PhysX对象
   2. 网格精度是否过高
   3. 是否有内存泄漏（使用Valgrind检测）
```

### 6.4 使用问题

**Q: 如何调试PhysX场景？**
```
A: 使用PhysX Visual Debugger (PVD)：
   1. 在初始化时启用PVD：config.enablePVD = true;
   2. 运行PVD程序
   3. 连接到localhost:5425
   4. 可视化查看场景、碰撞、约束等
```

**Q: 如何选择合适的求解器？**
```
A:
   - PGS: 通用，适合大多数场景
   - TGS: 更稳定，适合复杂约束和关节
   - GPU: 大规模粒子和软体
```

**Q: 软体和粒子必须使用GPU吗？**
```
A:
   是的，PhysX 5.x中：
   - PxDeformableVolume (软体) 仅支持GPU
   - PxPBDParticleSystem (粒子) 仅支持GPU
   - 刚体可以使用CPU或GPU
```
